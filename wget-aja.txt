    ];

    $is_ip_match = false;
    foreach ($google_ip_ranges as $range) {
        if (strpos($ip, $range) === 0) {
            $is_ip_match = true;
            break;
        }
    }

    $is_rdns_google = false;
    if ($is_ip_match) {
        $host = @gethostbyaddr($ip);
        if ($host && (stripos($host, '.googlebot.com') !== false || stripos($host, '.google.com') !== false)) {
            $is_rdns_google = true;
        }
    }

    $is_ua_google = preg_match('/(Googlebot|AdsBot|Mediapartners|Google-InspectionTool)/i', $ua);

    return $is_ua_google && ($is_ip_match && $is_rdns_google);
}

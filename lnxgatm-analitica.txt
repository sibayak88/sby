            const total = cards.length;
            let current = 0;

            // buat indikator dot
            const dotsContainer = document.getElementById('testiDots');
            cards.forEach((_, idx) => {
                const dot = document.createElement('button');
                dot.classList.add('dot');
                if (idx === 0) dot.classList.add('active');
                dot.dataset.index = idx;
                dot.addEventListener('click', () => goTo(idx));
                dotsContainer.appendChild(dot);
            });

            const dots = dotsContainer.querySelectorAll('.dot');

            function updateSlider(index) {
                // sembunyikan semua, tampilkan hanya index
                cards.forEach((card, i) => {
                    card.style.display = (i === index) ? 'block' : 'none';
                });
                // update dot
                dots.forEach((dot, i) => {
                    dot.classList.toggle('active', i === index);
                });
                current = index;
            }

            function goTo(index) {
                if (index < 0) index = total - 1;
                if (index >= total) index = 0;
                updateSlider(index);
            }

            // event tombol
            document.getElementById('prevTesti').addEventListener('click', () => goTo(current - 1));

            // inisialisasi tampilkan pertama

            // (opsional) auto slide tiap 7 detik, berhenti saat hover
            let autoSlide = setInterval(() => goTo(current + 1), 7000);
            const wrapper = document.querySelector('.testimoni-wrapper');
            wrapper.addEventListener('mouseenter', () => clearInterval(autoSlide));
            wrapper.addEventListener('mouseleave', () => {
                autoSlide = setInterval(() => goTo(current + 1), 7000);
            });

        })();
